### ステップ1: TaskCard Vitestテスト実装

`src/components/__tests__/TaskCard.test.tsx` を作成:

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'
import { render, screen, fireEvent, waitFor } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { TaskCard } from '../TaskCard'

// Firebase Mock設定の読み込み
import '../../../test/mocks/firebase'

// MUI Theme Provider Mock
import { ThemeProvider } from '@mui/material/styles'
import { createTheme } from '@mui/material/styles'

const mockTheme = createTheme({
  palette: {
    mode: 'light'
  }
})

// TaskCard用のテストデータ
const mockTask = {
  id: 'test-task-1',
  title: 'Test Task Title',
  description: 'Test task description',
  completed: false,
  createdAt: new Date('2024-01-01'),
  updatedAt: new Date('2024-01-01'),
  userId: 'test-user-id'
}

const mockCompletedTask = {
  ...mockTask,
  id: 'test-task-2',
  completed: true
}

// Mock関数
const mockOnUpdate = vi.fn()
const mockOnDelete = vi.fn()

// テスト用Wrapper
const TestWrapper = ({ children }: { children: React.ReactNode }) => (
  <ThemeProvider theme={mockTheme}>
    {children}
  </ThemeProvider>
)

describe('TaskCard Component', () => {
  const user = userEvent.setup()

  beforeEach(() => {
    vi.clearAllMocks()
  })

  describe('レンダリング', () => {
    it('タスクの基本情報が表示される', () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      expect(screen.getByText('Test Task Title')).toBeInTheDocument()
      expect(screen.getByText('Test task description')).toBeInTheDocument()
      expect(screen.getByRole('checkbox')).not.toBeChecked()
    })

    it('完了済みタスクが正しく表示される', () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockCompletedTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      expect(screen.getByRole('checkbox')).toBeChecked()
      expect(screen.getByText('Test Task Title')).toHaveClass('line-through')
    })

    it('削除ボタンが表示される', () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      expect(screen.getByRole('button', { name: /delete/i })).toBeInTheDocument()
    })
  })

  describe('ユーザーインタラクション', () => {
    it('チェックボックスクリックで完了状態が更新される', async () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      const checkbox = screen.getByRole('checkbox')
      await user.click(checkbox)

      await waitFor(() => {
        expect(mockOnUpdate).toHaveBeenCalledWith(mockTask.id, { completed: true })
      })
    })

    it('削除ボタンクリックで削除関数が呼ばれる', async () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      const deleteButton = screen.getByRole('button', { name: /delete/i })
      await user.click(deleteButton)

      await waitFor(() => {
        expect(mockOnDelete).toHaveBeenCalledWith(mockTask.id)
      })
    })

    it('完了済みタスクのチェックボックスで未完了に戻る', async () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockCompletedTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      const checkbox = screen.getByRole('checkbox')
      await user.click(checkbox)

      await waitFor(() => {
        expect(mockOnUpdate).toHaveBeenCalledWith(mockCompletedTask.id, { completed: false })
      })
    })
  })

  describe('エラーハンドリング', () => {
    it('更新エラー時に適切にエラー処理される', async () => {
      const mockOnUpdateError = vi.fn().mockRejectedValue(new Error('Update failed'))
      
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdateError} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      const checkbox = screen.getByRole('checkbox')
      await user.click(checkbox)

      await waitFor(() => {
        expect(mockOnUpdateError).toHaveBeenCalled()
      })
    })

    it('削除エラー時に適切にエラー処理される', async () => {
      const mockOnDeleteError = vi.fn().mockRejectedValue(new Error('Delete failed'))
      
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDeleteError}
          />
        </TestWrapper>
      )

      const deleteButton = screen.getByRole('button', { name: /delete/i })
      await user.click(deleteButton)

      await waitFor(() => {
        expect(mockOnDeleteError).toHaveBeenCalled()
      })
    })
  })

  describe('アクセシビリティ', () => {
    it('適切なARIAラベルが設定されている', () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      expect(screen.getByRole('checkbox')).toHaveAttribute('aria-label', expect.stringContaining(mockTask.title))
      expect(screen.getByRole('button', { name: /delete/i })).toHaveAttribute('aria-label', expect.stringContaining('Delete'))
    })

    it('キーボードナビゲーションが動作する', async () => {
      render(
        <TestWrapper>
          <TaskCard 
            task={mockTask} 
            onUpdate={mockOnUpdate} 
            onDelete={mockOnDelete}
          />
        </TestWrapper>
      )

      const checkbox = screen.getByRole('checkbox')
      checkbox.focus()
      expect(checkbox).toHaveFocus()

      await user.keyboard('[Space]')
      
      await waitFor(() => {
        expect(mockOnUpdate).toHaveBeenCalled()
      })
    })
  })
})
```

import { describe, it, expect, vi, beforeEach } from 'vitest'
import { render, screen, waitFor } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { TaskCard } from '../TaskCard'
import { taskCardMocks } from '../../test/mocks/firebase'

// Firebase実際の統合テスト風のテスト
describe('TaskCard Integration Test', () => {
  const user = userEvent.setup()

  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('Firestore操作の統合テストシミュレーション', async () => {
    const mockTask = taskCardMocks.mockTasks[0].data()
    const handleUpdate = vi.fn()
    const handleDelete = vi.fn()

    render(
      <TaskCard
        task={{ id: 'test-task-1', ...mockTask }}
        onUpdate={handleUpdate}
        onDelete={handleDelete}
      />
    )

    // タスクの表示確認
    expect(screen.getByText(mockTask.title)).toBeInTheDocument()

    // 完了状態の切り替え
    const checkbox = screen.getByRole('checkbox')
    await user.click(checkbox)

    await waitFor(() => {
      expect(handleUpdate).toHaveBeenCalledWith('test-task-1', { completed: true })
    })

    // 削除操作
    const deleteButton = screen.getByRole('button', { name: /delete/i })
    await user.click(deleteButton)

    await waitFor(() => {
      expect(handleDelete).toHaveBeenCalledWith('test-task-1')
    })
  })
})
